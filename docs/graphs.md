```mermaid
    curve: linear
---

	ingest_input([ingest_input]):::first
	ingest(ingest)
	extract(extract)
	write(write)
	factcheck(factcheck)
	flows(flows)
	render(render)
	render_output([render_output]):::last
	ingest_input --> ingest;
	ingest --> extract;
	extract --> write;
	write --> factcheck;
	factcheck --> flows;
	render --> render_output;
	flows --> render;

	classDef first fill-opacity:0
	classDef last fill:#bfb6fc
```
